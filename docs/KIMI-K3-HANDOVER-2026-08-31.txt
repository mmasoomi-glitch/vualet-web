/**
 * Encrypted-at-rest store for Baileys authentication state.
 *
 * Stores one row per session in the wa_auth_state table. This replaces the
 * old file-per-key layout, where nearly all of the auth state files were
 * unencrypted. loadAuthBlob is the only retrieval path and returns plaintext
 * to the caller; nothing else in this module may log or return the decrypted
 * payload, and plaintext/ciphertext must never appear in SQL, log lines, or
 * error messages.
 */

import { getDb, encrypt, decrypt } from '@mira/state';

export async function loadAuthBlob(tenantId, sessionId) {
  if (typeof tenantId !== 'string' || tenantId.length === 0) {
    throw new TypeError('tenantId must be a non-empty string');
  }

  if (typeof sessionId !== 'string' || sessionId.length === 0) {
    throw new TypeError('sessionId must be a non-empty string');
  }

  const { rows } = await getDb().query(
    'SELECT blob FROM wa_auth_state WHERE session_id = $1 AND tenant_id = $2',
    [sessionId, tenantId]
  );

  if (rows.length === 0) {
    return null;
  }

  return decrypt(tenantId, rows[0].blob);
}

export async function saveAuthBlob(tenantId, sessionId, serialized) {
  if (typeof tenantId !== 'string' || tenantId.length === 0) {
    throw new TypeError('tenantId must be a non-empty string');
  }

  if (typeof sessionId !== 'string' || sessionId.length === 0) {
    throw new TypeError('sessionId must be a non-empty string');
  }

  if (typeof serialized !== 'string') {
    throw new TypeError('serialized must be a string');
  }

  // CRITICAL: serialized is encrypted before it reaches the query. The
  // plaintext must never appear in SQL, log lines, or error messages.
  const encrypted = await encrypt(tenantId, serialized);

  await getDb().query(
    'INSERT INTO wa_auth_state (session_id, tenant_id, blob) VALUES ($1, $2, $3) ON CONFLICT (session_id) DO UPDATE SET blob = EXCLUDED.blob, tenant_id = EXCLUDED.tenant_id, updated_at = now()',
    [sessionId, tenantId, encrypted]
  );
}

export async function deleteAuthBlob(tenantId, sessionId) {
  if (typeof tenantId !== 'string' || tenantId.length === 0) {
    throw new TypeError('tenantId must be a non-empty string');
  }

  if (typeof sessionId !== 'string' || sessionId.length === 0) {
    throw new TypeError('sessionId must be a non-empty string');
  }

  const { rows } = await getDb().query(
    'DELETE FROM wa_auth_state WHERE session_id = $1 AND tenant_id = $2 RETURNING session_id',
    [sessionId, tenantId]
  );

  return rows.length;
}

/*
 * Drop-in replacement for useMultiFileAuthState.
 * The whole authentication state lives in one encrypted row rather than
 * thousands of plaintext files.
 * BufferJSON.replacer and BufferJSON.reviver must never be removed because
 * the state contains Buffers.

import { initAuthCreds, BufferJSON, proto } from '@whiskeysockets/baileys';
import { loadAuthBlob, saveAuthBlob } from './wa-auth-store.mjs';

export async function usePgAuthState(tenantId, sessionId) {
  let creds;
  let keys = {};

  const blob = await loadAuthBlob(tenantId, sessionId);

  if (typeof blob === 'string') {
    try {
      const parsed = JSON.parse(blob, BufferJSON.reviver);
      if (parsed && typeof parsed === 'object' && parsed.creds && typeof parsed.creds === 'object' && !Array.isArray(parsed.creds) && Object.keys(parsed.creds).length > 0) {
        creds = parsed.creds;
        keys = parsed.keys && typeof parsed.keys === 'object' && !Array.isArray(parsed.keys) ? parsed.keys : {};
      } else {
        creds = initAuthCreds();
        keys = {};
      }
    } catch (error) {
      creds = initAuthCreds();
      keys = {};
    }
  } else {
    creds = initAuthCreds();
    keys = {};
  }

  let writeChain = Promise.resolve();

  function serializeState() {
    return JSON.stringify({ creds, keys }, BufferJSON.replacer);
  }

  function persist() {
    const serialized = serializeState();
    writeChain = writeChain
      .catch(() => {})
      .then(() => saveAuthBlob(tenantId, sessionId, serialized));
    return writeChain;
  }

  async function saveCreds() {
    await persist();
  }

  const state = {
    creds,
    keys: {
      get(type, ids) {
        const result = {};
        if (!Array.isArray(ids)) {
          return result;
        }
        const typeStore = keys[type];
        if (!typeStore || typeof typeStore !== 'object') {
          return result;
        }
        for (const id of ids) {
          if (Object.prototype.hasOwnProperty.call(typeStore, id)) {
            const value = typeStore[id];
            if (type === 'app-state-sync-key') {
              result[id] = proto.Message.AppStateSyncKeyData.fromObject(value);
            } else {
              result[id] = value;
            }
          }
        }
        return result;
      },
      set(data) {
        if (!data || typeof data !== 'object') {
          return Promise.resolve();
        }
        for (const [type, entries] of Object.entries(data)) {
          if (!entries || typeof entries !== 'object') {
            continue;
          }
          if (!keys[type] || typeof keys[type] !== 'object' || Array.isArray(keys[type])) {
            keys[type] = {};
          }
          for (const [id, value] of Object.entries(entries)) {
            if (value === null || value === undefined) {
              delete keys[type][id];
            } else {
              keys[type][id] = value;
            }
          }
        }
        return persist();
      }
    }
  };

  return { state, saveCreds };
}

/**
 * Shard hashing core.
 *
 * Virtual nodes are used so that adding a shard moves only about
 * 1/N of future placements rather than remapping everyone as modulo
 * would. These functions are pure and deterministic across
 * processes and restarts.
 */
import { createHash } from 'node:crypto';

export function createRing(shards, opts = {}) {
  if (!Array.isArray(shards)) {
    throw new TypeError('shards must be an array');
  }
  if (shards.length === 0) {
    throw new TypeError('shards must not be empty');
  }

  const seen = new Set();
  for (const shard of shards) {
    if (typeof shard !== 'string' || shard.length === 0) {
      throw new TypeError('shards must contain only non-empty strings');
    }
    if (seen.has(shard)) {
      throw new TypeError('shards must not contain duplicate ids');
    }
    seen.add(shard);
  }

  const replicas = opts && opts.replicas !== undefined ? opts.replicas : 128;
  if (!Number.isInteger(replicas) || replicas < 1) {
    throw new TypeError('replicas must be a positive integer');
  }

  const ring = [];
  for (const shard of shards) {
    for (let i = 0; i < replicas; i++) {
      const digest = createHash('sha256').update(shard + '#' + i).digest('hex');
      const position = parseInt(digest.slice(0, 8), 16) >>> 0;
      ring.push(Object.freeze({ position, shard }));
    }
  }

  ring.sort((a, b) => a.position - b.position);
  Object.freeze(ring);

  return Object.freeze({
    shards: Object.freeze([...shards]),
    replicas,
    ring
  });
}

export function pickShard(ring, customerId) {
  if (
    typeof ring !== 'object' ||
    ring === null ||
    !Array.isArray(ring.shards) ||
    ring.shards.length === 0 ||
    !Array.isArray(ring.ring) ||
    ring.ring.length === 0 ||
    !Number.isInteger(ring.replicas) ||
    ring.replicas < 1
  ) {
    throw new TypeError('ring must be a ring object produced by createRing');
  }

  if (typeof customerId !== 'string' || customerId.length === 0) {
    throw new TypeError('customerId must be a non-empty string');
  }

  const digest = createHash('sha256').update(customerId).digest('hex');
  const position = parseInt(digest.slice(0, 8), 16) >>> 0;
  const nodes = ring.ring;

  let low = 0;
  let high = nodes.length - 1;

  if (position > nodes[high].position) {
    return nodes[0].shard;
  }

  while (low < high) {
    const mid = Math.floor((low + high) / 2);
    if (nodes[mid].position < position) {
      low = mid + 1;
    } else {
      high = mid;
    }
  }

  return nodes[low].shard;
}

export function assign(ring, customerId, currentAssignment) {
  // Existing placement always beats the hash: this stops a deploy or a shard
  // addition from dropping live customer connections. Anyone 'simplifying'
  // this into a bare pickShard call will silently disconnect every customer
  // whose hash has drifted.
  if (typeof currentAssignment === 'string' && currentAssignment.length > 0 && ring.shards.includes(currentAssignment)) {
    return currentAssignment;
  }
  return pickShard(ring, customerId);
}

export function planForNewShard(ring, nextRing, customerIds) {
  if (!Array.isArray(customerIds)) {
    throw new TypeError('customerIds must be an array');
  }
  const moved = [];
  const stayed = [];
  for (let i = 0; i < customerIds.length; i += 1) {
    const customerId = customerIds[i];
    const from = pickShard(ring, customerId);
    const to = pickShard(nextRing, customerId);
    if (from === to) {
      stayed.push(customerId);
    } else {
      moved.push({ customerId: customerId, from: from, to: to });
    }
  }
  return { moved: moved, stayed: stayed };
}

export function ringStats(ring, customerIds) {
  if (!Array.isArray(customerIds)) {
    throw new TypeError('customerIds must be an array');
  }
  const counts = {};
  for (let i = 0; i < ring.shards.length; i += 1) {
    counts[ring.shards[i]] = 0;
  }
  for (let i = 0; i < customerIds.length; i += 1) {
    const shard = pickShard(ring, customerIds[i]);
    counts[shard] = (counts[shard] || 0) + 1;
  }

  let min = 0;
  let max = 0;
  if (ring.shards.length > 0) {
    min = counts[ring.shards[0]];
    max = counts[ring.shards[0]];
    for (let i = 1; i < ring.shards.length; i += 1) {
      const count = counts[ring.shards[i]];
      if (count < min) {
        min = count;
      }
      if (count > max) {
        max = count;
      }
    }
  }

  return {
    counts: counts,
    min: min,
    max: max,
    spread: max - min,
    total: customerIds.length
  };
}
